DEFINITION L2R.Async.update := fun {n} {β} f c v d => if h : d = c then L2R.Async.update._proof_1 c d h ▸ v else f d
DEFINITION L2R.Async.initial := fun N => { state := N.init, inbox := fun x => [] }
DEFINITION L2R.Async.post := fun N k m => { state := k.state, inbox := L2R.Async.update k.inbox m.fst (k.inbox m.fst ++ [m.snd]) }
DEFINITION L2R.Async.emitted := fun N c x =>   match x with   | Option.none => []   | Option.some o => N.route c o
DEFINITION L2R.Async.deliver := fun N k c =>   match k.inbox c with   | [] => Option.none   | i :: rest =>     have r := N.step c (k.state c) i;     Option.some (List.foldl (L2R.Async.post N) { state := L2R.Async.update k.state c r.fst, inbox := L2R.Async.update k.inbox c rest } (L2R.Async.emitted N c r.snd))

DEFINITION L2R.Async.RoundRobin.order := fun N next => List.filterMap (fun off => if h : (next + off) % N.size < N.size then Option.some ⟨(next + off) % N.size, h⟩ else Option.none) (List.range N.size)
DEFINITION L2R.Async.RoundRobin.pushAll := fun N x x_1 => List.brecOn (motive := fun x => L2R.Async.Config N → L2R.Async.Config N) x_1 (L2R.Async.RoundRobin.pushAll._f N) x
DEFINITION L2R.Async.RoundRobin.serve := fun N k c =>   match k.inbox c with   | [] => Option.none   | input :: rest =>     match N.step c (k.state c) input with     | (state, output) =>       have k1 := { state := L2R.Async.update k.state c state, inbox := L2R.Async.update k.inbox c rest };       match output with       | Option.none => Option.some k1       | Option.some o => Option.some (L2R.Async.RoundRobin.pushAll N k1 (N.route c o))
DEFINITION L2R.Async.RoundRobin.stepOnce := fun N s => List.findSome? (fun c => Option.map (fun k => { cfg := k, next := (↑c + 1) % N.size }) (L2R.Async.RoundRobin.serve N s.cfg c)) (L2R.Async.RoundRobin.order N s.next)
DEFINITION L2R.Async.RoundRobin.send := fun N s c i => { cfg := { state := s.cfg.state, inbox := L2R.Async.update s.cfg.inbox c (s.cfg.inbox c ++ [i]) }, next := s.next }
